```python
from sqlalchemy import Column, String, Float, Text, Integer
from app.database import Base

class StoryNode(Base):
    __tablename__ = "story_nodes"

    id            = Column(String, primary_key=True)       # "A", "B", "S1" etc
    name          = Column(String, nullable=False)          # 显示名称
    position      = Column(Float, nullable=False)           # 0~200 环面位置
    node_type     = Column(String, nullable=False, default="normal")  # main|normal|special_shortcut|special_warp
    time_label    = Column(String, nullable=True)           # "第一天·傍晚 18:30"
    content       = Column(Text, nullable=False)            # 正文（Markdown）
    speaker       = Column(String, nullable=True)           # null=旁白, npc_id=对话
    background    = Column(String, nullable=True)           # 背景图路径

    # 扩展字段（JSON 字符串）
    cycle_variants_json = Column(Text, default="{}")        # 循环变体
    metadata_json       = Column(Text, default="{}")        # color_palette, atmosphere, sensory
    parent_node_id      = Column(String, nullable=True)     # 子节点关联的主节点
```

### 1.2 Pydantic Schema 实现（Day 4-5）

#### 1.2.1 核心 Schema 层级

```
BaseSchema (id, 通用字段)
  ├── NodeSchema (完整节点信息)
  ├── ChoiceSchema (选择分支)
  ├── ItemSchema (道具)
  ├── GameState (运行时状态)
  ├── Frame (一帧游戏画面，含 node + state + choices + persistent)
  └── CycleEvent (循环完成事件)
```

#### 1.2.2 关键 Schema 定义

```python
# backend/app/schemas/game.py
from pydantic import BaseModel
from typing import Optional
from enum import Enum

class NodeType(str, Enum):
    MAIN = "main"
    NORMAL = "normal"
    SPECIAL_SHORTCUT = "special_shortcut"
    SPECIAL_WARP = "special_warp"

class ItemBrief(BaseModel):
    """Frame 中展示的道具简要信息"""
    id: str
    name: str
    cross_surface: bool = False

class ChoiceResult(BaseModel):
    """前端渲染用的选项"""
    id: str
    text: str
    available: bool
    source: str = "static"               # static|special_shortcut|special_warp
    reason: Optional[str] = None         # 不可用时的原因

class GameState(BaseModel):
    """游戏运行时状态"""
    current_node_id: str
    cycle_count: int = 0
    half_cycle_count: int = 0
    inventory: list[ItemBrief] = []
    flags: dict[str, any] = {}
    visited_nodes: list[str] = []
    player_attributes: dict[str, int] = {}
    endings_reached: list[str] = []

class PersistentFound(BaseModel):
    """在节点发现的遗留物"""
    items: list[ItemBrief] = []
    cross_surface_items: list[ItemBrief] = []
    dangers: list[dict] = []

class CycleEvent(BaseModel):
    """循环完成事件"""
    is_cycle_end: bool = True
    is_full_cycle: bool = True           # True=完整圈, False=捷径半循环
    ending_text: Optional[str] = None    # 结局描述文本
    total_cycles: int = 0
    total_half_cycles: int = 0

class Frame(BaseModel):
    """一帧游戏画面"""
    node: NodeSchema
    state: GameState
    available_choices: list[ChoiceResult]
    persistent_found: PersistentFound = PersistentFound()
    cycle_event: Optional[CycleEvent] = None
```

#### 1.3.2 种子数据脚本结构

```python
# backend/seed_data.py
async def seed_demo_nodes(session):
    """创建 8 主节点 + 2 特殊节点的最小 demo 数据"""
    nodes = [
        StoryNode(id="A", name="荔湾广场正门", position=0.0,  node_type="main",
                  content="你站在荔湾广场正门前...", time_label="第一天·傍晚 18:30"),
        StoryNode(id="B", name="德星路出租屋", position=25.0, node_type="main",
                  content="你推开了502的房门...", time_label="第一天·深夜 23:00"),
        # ... C, D, E, F, G, H
        StoryNode(id="J", name="地宫暗道", position=-1.0, node_type="special_shortcut",
                  content="你弯腰钻进了狭窄的密道..."),
        StoryNode(id="K", name="龙脉裂隙", position=-1.0, node_type="special_warp",
                  content="灰白色的虚空在你脚下展开..."),
    ]
    session.add_all(nodes)

    choices = [
        Choice(id="A_choice_01", from_node_id="A", text="进入广场", next_node_id="B",
               effects=[{"type": "set_flag", "target": "started_demo", "value": True}],
               priority=1),
        # ... 形成 A→B→C→D→E→F→G→H→A 的完整圈
    ]
    session.add_all(choices)
    await session.commit()
```

#### 1.4.2 校验规则

| 规则 | 说明 |
|------|------|
| 外键完整性 | 所有 `next_node_id` 必须指向已定义的 `story_nodes.id` |
