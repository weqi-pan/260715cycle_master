```python
# backend/app/engine/graph.py
from app.models.story import StoryNode, Choice
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select

class Graph:
    """故事图结构 — 只负责查询，不负责状态修改"""

    def __init__(self, session: AsyncSession):
        self.session = session

    async def get_node(self, node_id: str) -> StoryNode | None:
        result = await self.session.execute(
            select(StoryNode).where(StoryNode.id == node_id)
        )
        return result.scalar_one_or_none()

    async def get_choices(self, node_id: str) -> list[Choice]:
        result = await self.session.execute(
            select(Choice)
            .where(Choice.from_node_id == node_id)
            .order_by(Choice.priority)
        )
        return list(result.scalars().all())

    async def get_all_main_nodes(self) -> list[StoryNode]:
        result = await self.session.execute(
            select(StoryNode)
            .where(StoryNode.node_type == "main")
            .order_by(StoryNode.position)
        )
        return list(result.scalars().all())

    async def get_special_node(self, node_type: str) -> StoryNode | None:
        result = await self.session.execute(
            select(StoryNode).where(StoryNode.node_type == node_type)
        )
        return result.scalar_one_or_none()

    async def node_exists(self, node_id: str) -> bool:
        node = await self.get_node(node_id)
        return node is not None
```

### 2.2 condition_eval.py — 条件表达式解析器（Day 4-7）

#### 2.2.1 条件语法

```
基本条件：
  has_item:道具ID          背包中是否持有某道具
  has_flag:标记名          是否存在某标记
  flag:标记名=值           标记是否等于指定值
  attr:属性名>=值          属性比较 (>=, <=, >, <, ==, !=)
  cycle>=N                 完整循环次数
  half_cycle>=N            半循环次数
  at_node:节点ID           当前是否在某节点

组合条件：
  not:子条件               取反
  and:条件1,条件2,...      全部满足
  or:条件1,条件2,...       满足其一

null = 始终满足
```

#### 2.2.2 解析器实现

```python
# backend/app/engine/condition_eval.py
import re
from app.schemas.game import GameState

class ConditionEvaluator:
    """条件表达式解析器"""

    # 正则模式
    PATTERN_HAS_ITEM   = re.compile(r"^has_item:(.+)$")
    PATTERN_HAS_FLAG   = re.compile(r"^has_flag:(.+)$")
    PATTERN_FLAG_EQ    = re.compile(r"^flag:(.+)=(.+)$")
    PATTERN_ATTR       = re.compile(r"^attr:(.+)(>=|<=|>|<|==|!=)(.+)$")
    PATTERN_CYCLE      = re.compile(r"^cycle(>=|<=|>|<|==)(\d+)$")
    PATTERN_HALF_CYCLE = re.compile(r"^half_cycle(>=|<=|>|<|==)(\d+)$")
    PATTERN_AT_NODE    = re.compile(r"^at_node:(.+)$")
    PATTERN_NOT        = re.compile(r"^not:(.+)$")
    PATTERN_AND        = re.compile(r"^and:(.+)$")
    PATTERN_OR         = re.compile(r"^or:(.+)$")

    def __init__(self, state: GameState):
        self.state = state

    def evaluate(self, condition: str | None) -> bool:
        """主入口：求值条件表达式"""
        if condition is None:
            return True  # null = 始终满足

        condition = condition.strip()

        # 基本条件
        if m := self.PATTERN_HAS_ITEM.match(condition):
            return self._eval_has_item(m.group(1))
        if m := self.PATTERN_HAS_FLAG.match(condition):
            return self._eval_has_flag(m.group(1))
        if m := self.PATTERN_FLAG_EQ.match(condition):
            return self._eval_flag_eq(m.group(1), m.group(2))
        if m := self.PATTERN_ATTR.match(condition):
            return self._eval_attr(m.group(1), m.group(2), m.group(3))
        if m := self.PATTERN_CYCLE.match(condition):
            return self._eval_cycle(m.group(1), int(m.group(2)))
        if m := self.PATTERN_HALF_CYCLE.match(condition):
            return self._eval_half_cycle(m.group(1), int(m.group(2)))
        if m := self.PATTERN_AT_NODE.match(condition):
            return self._eval_at_node(m.group(1))

        # 组合条件
        if m := self.PATTERN_NOT.match(condition):
            return not self.evaluate(m.group(1))
        if m := self.PATTERN_AND.match(condition):
            sub_conditions = self._split_comma(m.group(1))
            return all(self.evaluate(sub) for sub in sub_conditions)
        if m := self.PATTERN_OR.match(condition):
            sub_conditions = self._split_comma(m.group(1))
            return any(self.evaluate(sub) for sub in sub_conditions)

        raise ValueError(f"Unknown condition syntax: {condition}")

    def _eval_has_item(self, item_id: str) -> bool:
        return any(item.id == item_id for item in self.state.inventory)

    def _eval_has_flag(self, flag_name: str) -> bool:
        return self.state.flags.get(flag_name) is not None

    def _eval_flag_eq(self, flag_name: str, value: str) -> bool:
        actual = self.state.flags.get(flag_name)
        return str(actual) == value

    def _eval_attr(self, attr_name: str, op: str, value: str) -> bool:
        actual = self.state.player_attributes.get(attr_name, 0)
        target = float(value)
        return self._compare(actual, op, target)

    def _eval_cycle(self, op: str, value: int) -> bool:
        return self._compare(self.state.cycle_count, op, value)

    def _eval_half_cycle(self, op: str, value: int) -> bool:
        return self._compare(self.state.half_cycle_count, op, value)

    def _eval_at_node(self, node_id: str) -> bool:
        return self.state.current_node_id == node_id

    @staticmethod
    def _compare(a, op: str, b) -> bool:
        if op == ">=": return a >= b
        if op == "<=": return a <= b
        if op == ">":  return a > b
        if op == "<":  return a < b
        if op == "==": return a == b
        if op == "!=": return a != b
        return False

    @staticmethod
    def _split_comma(text: str) -> list[str]:
        """按逗号分割，但保护嵌套的 and:/or: 内部"""
        parts = []
        depth = 0
        current = ""
        for char in text:
            if char == "," and depth == 0:
                parts.append(current.strip())
                current = ""
            else:
                if char in "(": depth += 1
                elif char in ")": depth -= 1
                current += char
        if current.strip():
            parts.append(current.strip())
        return parts

    @classmethod
    def check_syntax(cls, condition: str) -> tuple[bool, str | None]:
        """语法检查（不依赖 GameState）"""
        if condition is None:
            return (True, None)
        # Phase 2: 用试探性解析检查语法是否正确
        # Phase 6: 扩展为完整的静态语法校验
        try:
            condition = condition.strip()
            if cls.PATTERN_HAS_ITEM.match(condition): return (True, None)
            if cls.PATTERN_HAS_FLAG.match(condition): return (True, None)
            if cls.PATTERN_FLAG_EQ.match(condition): return (True, None)
            if cls.PATTERN_ATTR.match(condition): return (True, None)
            if cls.PATTERN_CYCLE.match(condition): return (True, None)
            if cls.PATTERN_HALF_CYCLE.match(condition): return (True, None)
            if cls.PATTERN_AT_NODE.match(condition): return (True, None)
            if cls.PATTERN_NOT.match(condition):
                return cls.check_syntax(cls.PATTERN_NOT.match(condition).group(1))
            if cls.PATTERN_AND.match(condition):
                subs = cls._split_comma(None, cls.PATTERN_AND.match(condition).group(1))
                for sub in subs:
                    ok, err = cls.check_syntax(sub)
                    if not ok: return (False, f"in and(): {err}")
                return (True, None)
            if cls.PATTERN_OR.match(condition):
                subs = cls._split_comma(None, cls.PATTERN_OR.match(condition).group(1))
                for sub in subs:
                    ok, err = cls.check_syntax(sub)
                    if not ok: return (False, f"in or(): {err}")
                return (True, None)
            return (False, f"Unknown condition: {condition}")
        except Exception as e:
            return (False, str(e))
```

### 2.3 special_router.py — 特殊节点路由（Day 8-10）

#### 2.3.1 功能需求

```
SpecialRouter
├── get_available(node, state) → list[ChoiceResult]
│   ├── K 跃迁入口：检查 entry_condition
│   ├── K 跃迁出口：在 K 节点时返回跳转选项
│   └── J 捷径入口：在 E 节点时检查 J 的触发条件
│
└── resolve(choice_id, node, state) → Choice
    ├── __warp_K_enter  → 进入 K 节点
    ├── __warp_K_exit_X → 从 K 跳转到节点 X
    └── __shortcut_J    → 进入 J 节点
```

#### 2.3.2 实现

```python
# backend/app/engine/special_router.py
from app.models.story import StoryNode, Choice
from app.schemas.game import GameState, ChoiceResult
from app.engine.condition_eval import ConditionEvaluator
from app.engine.graph import Graph

class SpecialRouter:
    """特殊节点（K 跃迁、J 捷径）的动态路由"""

    # 选项 ID 前缀约定
    WARP_ENTER_PREFIX  = "__warp_K_enter"
    WARP_EXIT_PREFIX   = "__warp_K_exit_"
    SHORTCUT_PREFIX    = "__shortcut_J"

    def __init__(self, graph: Graph):
        self.graph = graph

    async def get_available(
        self, node: StoryNode, state: GameState
    ) -> list[ChoiceResult]:
        results = []
        evaluator = ConditionEvaluator(state)

        # K 跃迁入口：不在 K 节点时检查
        warp_node = await self.graph.get_special_node("special_warp")
        if warp_node and node.id != warp_node.id:
            entry_condition = warp_node.metadata.get("entry_condition")
            if entry_condition and evaluator.evaluate(entry_condition):
                results.append(ChoiceResult(
                    id=self.WARP_ENTER_PREFIX,
                    text="踏入突然出现的裂隙",
                    available=True,
                    source="special_warp"
                ))

        # K 跃迁出口：在 K 节点时生成所有主节点跳转选项
        if node.id == "K":
            main_nodes = await self.graph.get_all_main_nodes()
            for target in main_nodes:
                results.append(ChoiceResult(
                    id=f"{self.WARP_EXIT_PREFIX}{target.id}",
                    text=f"跃迁至 {target.name}",
                    available=True,
                    source="special_warp"
                ))

        # J 捷径入口：在 E 节点时检查
        if node.id == "E":
            j_node = await self.graph.get_special_node("special_shortcut")
            if j_node:
                # J 的进入条件来自 E→J 的 choice
                e_choices = await self.graph.get_choices("E")
                j_choice = next(
                    (c for c in e_choices if c.next_node_id == "J"), None
                )
                if j_choice and evaluator.evaluate(j_choice.condition):
                    results.append(ChoiceResult(
                        id=self.SHORTCUT_PREFIX,
                        text="走捷径（莫比乌斯折返）",
                        available=True,
                        source="special_shortcut"
                    ))

        return results

    def resolve(self, choice_id: str, from_node_id: str) -> Choice:
        """将特殊选项 ID 解析为伪 Choice 对象"""
        if choice_id == self.WARP_ENTER_PREFIX:
            return Choice(
                id=choice_id, from_node_id=from_node_id,
                text="", next_node_id="K",
                condition=None, effects=[
                    {"type": "damage", "target": "sanity", "value": "san_max-1"}
                ]
            )
        elif choice_id.startswith(self.WARP_EXIT_PREFIX):
            target_id = choice_id[len(self.WARP_EXIT_PREFIX):]
            return Choice(
                id=choice_id, from_node_id="K",
                text="", next_node_id=target_id,
                condition=None, effects=[]
            )
        elif choice_id == self.SHORTCUT_PREFIX:
            return Choice(
                id=choice_id, from_node_id="E",
                text="", next_node_id="J",
                condition=None, effects=[]
            )
        else:
            raise ValueError(f"Unknown special choice: {choice_id}")
```

### 2.4 engine.py — 游戏主循环（Day 11-15）

#### 2.4.1 核心流程

```
process_choice(node_id, choice_id, save_id) → Frame

① 加载当前节点和 state
② 条件校验（静态选择 + 特殊路由）
③ 处理静态选择 vs 特殊选择的分派
④ 应用 Effects（修改 state）
⑤ 特殊路由处理 → 目标节点
⑥ 循环检测（回到 A 时触发 cycle_event）
⑦ 加载节点遗留状态（含跨面道具）
⑧ 保存 state 到数据库
⑨ 解析目标节点的可用选项
⑩ 组装 Frame 返回
```

#### 2.4.2 效果应用器

```python
# backend/app/engine/effects.py
from app.schemas.game import GameState, ItemBrief

class EffectApplier:
    """将 Effect 列表应用到 GameState"""

    @staticmethod
    def apply(state: GameState, effect: dict) -> GameState:
        etype  = effect["type"]
        target = effect["target"]
        value  = effect.get("value")

        if etype == "add_item":
            # 添加道具到背包（去重）
            if not any(i.id == target for i in state.inventory):
                state.inventory.append(ItemBrief(
                    id=target, name=target, cross_surface=False  # name 从 DB 加载
                ))

        elif etype == "remove_item":
            state.inventory = [i for i in state.inventory if i.id != target]

        elif etype == "set_flag":
            state.flags[target] = value if value is not None else True

        elif etype == "remove_flag":
            state.flags.pop(target, None)

        elif etype == "toggle_flag":
            current = state.flags.get(target, False)
            state.flags[target] = not current

        elif etype == "heal":
            if target in state.player_attributes:
                state.player_attributes[target] += int(value)

        elif etype == "damage":
            if target in state.player_attributes:
                state.player_attributes[target] -= int(value)

        elif etype == "set_attr":
            state.player_attributes[target] = int(value)

        elif etype == "leave_item":
            # 遗留道具在当前节点（由 engine 层处理，通过 node_persistent_state）
            pass  # 标记后在 engine.process_choice() 中处理

        elif etype == "leave_danger":
            # 同上
            pass

        elif etype == "set_san_max":
            # 修改 san 值上限
            # 需要感知 max_value（从 player_attributes_def 获取）
            pass

        return state

    @staticmethod
    def apply_all(state: GameState, effects: list[dict]) -> GameState:
        for effect in effects:
            state = EffectApplier.apply(state, effect)
        return state
```

#### 2.4.3 引擎主函数

```python
# backend/app/engine/engine.py
async def process_choice(
    session: AsyncSession,
    node_id: str,
    choice_id: str,
    save_id: str
) -> Frame:
    """游戏主循环核心 — 处理一次选择，返回下一帧"""

    graph = Graph(session)
    router = SpecialRouter(graph)
    state = await load_state(session, save_id)
    node = await graph.get_node(node_id)

    if node is None:
        raise ValueError(f"Node not found: {node_id}")

    # ① 获取选择对象
    if choice_id.startswith("__"):
        # 特殊路由选项
        choice = router.resolve(choice_id, node_id)
    else:
        choice = await graph.get_choice(node_id, choice_id)
        if choice is None:
            raise ValueError(f"Choice not found: {choice_id}")

    # ② 条件校验
    evaluator = ConditionEvaluator(state)
    if choice.condition and not evaluator.evaluate(choice.condition):
        raise ConditionNotMetError(choice.condition)

    # ③ 应用 Effects
    effects = json.loads(choice.effects) if isinstance(choice.effects, str) else choice.effects
    state = EffectApplier.apply_all(state, effects)

    # ④ 获取目标节点
    next_node = await graph.get_node(choice.next_node_id)
    if next_node is None:
        raise ValueError(f"Target node not found: {choice.next_node_id}")

    # ⑤ 循环检测
    cycle_event = None
    if next_node.id == "A" and len(state.visited_nodes) > 0:
        if _came_via_shortcut(choice_id, node_id):
            state.half_cycle_count += 1
            is_full = False
        else:
            state.cycle_count += 1
            is_full = True

        state.visited_nodes = []
        cycle_event = CycleEvent(
            is_cycle_end=True,
            is_full_cycle=is_full,
            ending_text=generate_ending_text(state),
            total_cycles=state.cycle_count,
            total_half_cycles=state.half_cycle_count
        )

    # ⑥ 更新 visited_nodes
    state.visited_nodes.append(next_node.id)
    state.current_node_id = next_node.id

    # ⑦ 加载节点遗留状态（含跨面道具）
    persistent = await load_persistent_state(session, save_id, next_node.id)
    if next_node.id in ("A", "E"):
        mirror_id = "E" if next_node.id == "A" else "A"
        mirror_state = await load_persistent_state(session, save_id, mirror_id)
        persistent.cross_surface_items = [
            item for item in mirror_state.items if item.cross_surface
        ]

    # ⑧ 保存 state
    await save_state(session, save_id, state)

    # ⑨ 解析可用选项（静态 + 特殊）
    static_choices = await graph.get_choices(next_node.id)
    formatted_static = []
    for c in static_choices:
        available = True
        reason = None
        if c.condition:
            available = evaluator.evaluate(c.condition)
            if not available:
                reason = f"需要: {c.condition}" if not c.is_hidden_when_locked else None

        if not c.is_hidden_when_locked or available:
            formatted_static.append(ChoiceResult(
                id=c.id, text=c.text, available=available,
                source="static", reason=reason
            ))

    special_choices = await router.get_available(next_node, state)
    all_choices = formatted_static + special_choices

    # ⑩ 组装 Frame
    return Frame(
        node=node_to_schema(next_node),
        state=state,
        available_choices=all_choices,
        persistent_found=persistent,
        cycle_event=cycle_event
    )


def _came_via_shortcut(choice_id: str, from_node_id: str) -> bool:
    """判断是否通过 J 捷径到达 A"""
    return choice_id == "J_choice_exit" or from_node_id == "J"


def generate_ending_text(state: GameState) -> str:
    """根据当前状态生成结局描述"""
    # Phase 2: 简单版本
    cycle = state.cycle_count
    half = state.half_cycle_count
    courage = state.player_attributes.get("courage", 0)

    if cycle == 0 and half == 1:
        return "你通过密道回到了起点。天还没亮——这一次，你比时间快了一步。"
    elif cycle == 1:
        return "你完成了第一次循环。夕阳依旧染红了穹顶，但你已经不一样了。"
    elif cycle >= 8:
        return "你已经记不清走了多少圈了。莫比乌斯环没有终点——但也许，你开始不想找到终点了。"
    elif courage >= 8:
        return "循环在继续，但你已经不再害怕了。不是麻木——是你选择了面对。"
    else:
        return f"第{cycle}次回到起点。故事还在继续。"
```
