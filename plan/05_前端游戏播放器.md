### 2.1 GameStore — Pinia 状态管理（Day 1-3）

```typescript
// frontend/src/stores/gameStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { Frame, GameState, StoryNode, ChoiceResult, ItemBrief } from '@/types'
import { gameApi, savesApi } from '@/api'

export const useGameStore = defineStore('game', () => {
  // ====== 状态 ======
  const currentNode = ref<StoryNode | null>(null)
  const gameState = ref<GameState | null>(null)
  const availableChoices = ref<ChoiceResult[]>([])
  const persistentFound = ref<{ items: ItemBrief[]; crossSurfaceItems: ItemBrief[]; dangers: any[] } | null>(null)
  const cycleEvent = ref<any>(null)
  const saveId = ref<string | null>(null)

  const isLoading = ref(false)
  const isTypingComplete = ref(false)      // 打字机效果是否完成
  const showCycleEndOverlay = ref(false)   // 是否显示循环结束动画

  // ====== 计算属性 ======
  const currentCycleCount = computed(() => gameState.value?.cycleCount ?? 0)
  const currentHalfCycleCount = computed(() => gameState.value?.halfCycleCount ?? 0)
  const inventory = computed(() => gameState.value?.inventory ?? [])
  const playerAttributes = computed(() => gameState.value?.playerAttributes ?? {})
  const hasCrossSurfaceItems = computed(() =>
    (persistentFound.value?.crossSurfaceItems?.length ?? 0) > 0
  )

  // ====== 动作 ======
  async function startNewGame() {
    isLoading.value = true
    try {
      const frame: Frame = await gameApi.start()
      applyFrame(frame)
    } finally {
      isLoading.value = false
    }
  }

  async function makeChoice(choiceId: string) {
    if (!currentNode.value || !saveId.value) return

    isLoading.value = true
    isTypingComplete.value = false
    try {
      const frame: Frame = await gameApi.choose(
        currentNode.value.id, choiceId, saveId.value
      )
      applyFrame(frame)

      // 循环结束时触发过渡动画
      if (frame.cycleEvent) {
        showCycleEndOverlay.value = true
        // 2 秒后自动关闭
        setTimeout(() => { showCycleEndOverlay.value = false }, 2500)
      }
    } catch (err) {
      console.error('Choice failed:', err)
    } finally {
      isLoading.value = false
    }
  }

  function applyFrame(frame: Frame) {
    currentNode.value = frame.node
    gameState.value = frame.state
    availableChoices.value = frame.availableChoices
    persistentFound.value = frame.persistentFound
    cycleEvent.value = frame.cycleEvent
  }

  function onTypingComplete() {
    isTypingComplete.value = true
  }

  // 存档相关
  async function loadSave(saveSlotId: string) {
    // 从存档恢复游戏
  }

  async function saveGame(saveName?: string) {
    // 手动保存
  }

  return {
    // state
    currentNode, gameState, availableChoices, persistentFound, cycleEvent,
    saveId, isLoading, isTypingComplete, showCycleEndOverlay,
    // computed
    currentCycleCount, currentHalfCycleCount, inventory, playerAttributes,
    hasCrossSurfaceItems,
    // actions
    startNewGame, makeChoice, applyFrame, onTypingComplete,
    loadSave, saveGame,
  }
})
```

### 2.3 TypewriterText.vue — 打字机效果（Day 6-8）

#### 2.3.1 实现

```vue
<!-- frontend/src/components/player/TypewriterText.vue -->
<template>
  <div class="typewriter-text" ref="containerRef">
    <div class="text-content" v-html="renderedText"></div>
    <span v-if="!isComplete" class="cursor">▌</span>
  </div>
</template>

<script setup lang="ts">
import { ref, watch, onUnmounted } from 'vue'

const props = defineProps<{
  content: string
  speed?: number       // 字/秒，默认 45
  pauseOnPunctuation?: boolean
}>()

const emit = defineEmits<{ typingComplete: [] }>()

const renderedText = ref('')
const isComplete = ref(false)
let timer: number | null = null

function startTyping(text: string) {
  renderedText.value = ''
  isComplete.value = false
  let index = 0
  const chars = [...text]  // 正确处理 Unicode/emoji
  const speed = props.speed ?? 45
  const interval = 1000 / speed

  timer = window.setInterval(() => {
    if (index >= chars.length) {
      isComplete.value = true
      emit('typingComplete')
      if (timer) clearInterval(timer)
      return
    }

    let char = chars[index]
    renderedText.value += char
    index++

    // 标点符号额外停顿
    if (props.pauseOnPunctuation !== false && '。！？；：，、\n'.includes(char)) {
      if (timer) clearInterval(timer)
      const pause = char === '\n' ? 400 : 200
      timer = window.setTimeout(() => {
        startTyping(text.slice(index))
      }, pause)
    }
  }, interval)
}

watch(() => props.content, (newContent) => {
  if (timer) clearInterval(timer)
  startTyping(newContent)
}, { immediate: true })

onUnmounted(() => {
  if (timer) clearInterval(timer)
})

// 支持点击快进
function skip() {
  if (timer) clearInterval(timer)
  renderedText.value = props.content
  isComplete.value = true
  emit('typingComplete')
}
</script>
```

#### 2.3.2 Markdown 渲染支持

内容中的 Markdown 语法：
- `**粗体**` → `<strong>`
- `*斜体*` → `<em>`
- `---` → 暂停 500ms
- `> 引用` → `<blockquote>`
- `{{变量}}` → 替换为实际值（`cycle_count` 等）

使用一个轻量的 Markdown 渲染函数（不用外部库，因为只支持子集）：

```typescript
// frontend/src/utils/markdown.ts
export function renderMarkdownSubset(text: string, variables: Record<string, any> = {}): string {
  // 1. 替换模板变量
  let result = text.replace(/\{\{(.+?)\}\}/g, (_, key) => {
    return variables[key.trim()] ?? `{{${key}}}`
  })

  // 2. Markdown 子集转换
  result = result.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
  result = result.replace(/\*(.+?)\*/g, '<em>$1</em>')
  result = result.replace(/~~(.+?)~~/g, '<del>$1</del>')
  result = result.replace(/^> (.+)$/gm, '<blockquote>$1</blockquote>')
  result = result.replace(/^---$/gm, '<span class="scene-break"></span>')
  result = result.replace(/\\n/g, '<br>')

  return result
}
```

### 2.5 CycleMap.vue — 莫比乌斯环 SVG 小地图（Day 11-14）

#### 2.5.1 SVG 设计

```
                    莫比乌斯环（200单位，8个主节点 + 2个特殊节点）

                         B(25)
                    ___---  ---___
               C(50) /              \ D(75)
                    |   ╭──────╮   |
                    |   │ 环面 │   |
        J(捷径) ●───|   │ 扭转 │   |───● K(跃迁)
                    |   │ 180° │   |
               H(175)\             / E(100)
                     \___     ___/
                    G(150)    F(125)
                          A(0)
                   [莫比乌斯环中心]
```

#### 2.5.2 实现要点

```typescript
// frontend/src/components/player/CycleMap.vue
// 使用 SVG polar coordinates 放置节点

interface MapNode {
  id: string
  name: string
  position: number     // 0-200 环面位置
  type: 'main' | 'special_shortcut' | 'special_warp'
  isCurrent: boolean
  isVisited: boolean
}

// 环面参数
const RING_RADIUS = 120       // 主环半径
const CENTER_X = 150
const CENTER_Y = 150

function getNodePosition(position: number): { x: number; y: number } {
  // position 0 位于环面右下角（莫比乌斯环的"A面"）
  // E(100) 位于 A(0) 的正下方，表示"另一面"
  const angle = (position / 200) * 360 - 90  // -90° 起始
  const rad = (angle * Math.PI) / 180
  return {
    x: CENTER_X + RING_RADIUS * Math.cos(rad),
    y: CENTER_Y + RING_RADIUS * Math.sin(rad),
  }
}
```

**SVG 元素**：
- 灰色虚线环（基础环）
- 8 个主节点圆点（颜色区分：金色=当前位置、灰色=已访问、暗灰=未访问）
- 1 条莫比乌斯扭转虚线（A ↔ E 连线，表示"跨面"）
- 2 个特殊节点菱形标记（J 绿色、K 紫色）
- 节点标签（A~H 字母）
- 当前位置脉冲动画

### 2.9 API 对接层（Day 19-20）

```typescript
// frontend/src/api/game.ts
import apiClient from './client'
import type { Frame } from '@/types'

export const gameApi = {
  async start(): Promise<Frame> {
    return apiClient.get('/game/start') as Promise<Frame>
  },

  async choose(nodeId: string, choiceId: string, saveId: string): Promise<Frame> {
    return apiClient.post(`/game/choose/${nodeId}`, {
      choice_id: choiceId,
      save_id: saveId,
    }) as Promise<Frame>
  },

  async getState(saveId: string) {
    return apiClient.get(`/game/state/${saveId}`)
  },
}

// frontend/src/api/saves.ts
import apiClient from './client'
import type { SaveSlot } from '@/types'

export const savesApi = {
  async list(): Promise<SaveSlot[]> {
    return apiClient.get('/saves') as Promise<SaveSlot[]>
  },

  async create(saveName: string): Promise<SaveSlot> {
    return apiClient.post('/saves', { save_name: saveName }) as Promise<SaveSlot>
  },

  async update(saveId: string, data: any) {
    return apiClient.put(`/saves/${saveId}`, data)
  },

  async delete(saveId: string) {
    return apiClient.delete(`/saves/${saveId}`)
  },
}
```
