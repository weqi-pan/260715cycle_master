### 2.1 EditorStore — Pinia 编辑器状态（Day 1-2）

```typescript
// frontend/src/stores/editorStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { StoryNode, Choice, NodeCreate, ChoiceCreate } from '@/types'
import { editorApi } from '@/api'

export const useEditorStore = defineStore('editor', () => {
  // ====== 状态 ======
  const nodes = ref<StoryNode[]>([])
  const choices = ref<Choice[]>([])
  const selectedNodeId = ref<string | null>(null)
  const selectedChoiceId = ref<string | null>(null)
  const isDirty = ref(false)           // 是否有未保存修改

  // ====== 计算属性 ======
  const selectedNode = computed(() =>
    nodes.value.find(n => n.id === selectedNodeId.value) ?? null
  )
  const nodeChoices = computed(() =>
    choices.value.filter(c => c.fromNodeId === selectedNodeId.value)
  )
  const nodesByType = computed(() => ({
    main: nodes.value.filter(n => n.nodeType === 'main'),
    normal: nodes.value.filter(n => n.nodeType === 'normal'),
    special: nodes.value.filter(n =>
      n.nodeType === 'special_shortcut' || n.nodeType === 'special_warp'
    ),
  }))

  // ====== 动作 ======
  async function loadAllNodes() {
    nodes.value = await editorApi.getAllNodes()
    choices.value = await editorApi.getAllChoices()
    isDirty.value = false
  }

  function selectNode(nodeId: string | null) {
    selectedNodeId.value = nodeId
    selectedChoiceId.value = null
  }

  function selectChoice(choiceId: string | null) {
    selectedChoiceId.value = choiceId
  }

  async function saveNode(nodeData: NodeCreate | StoryNode) {
    if ('id' in nodeData && nodes.value.find(n => n.id === nodeData.id)) {
      await editorApi.updateNode(nodeData.id, nodeData)
    } else {
      await editorApi.createNode(nodeData)
    }
    await loadAllNodes()
    isDirty.value = false
  }

  async function deleteNode(nodeId: string) {
    await editorApi.deleteNode(nodeId)
    await loadAllNodes()
    if (selectedNodeId.value === nodeId) {
      selectedNodeId.value = null
    }
  }

  async function saveChoice(choiceData: ChoiceCreate) {
    await editorApi.createChoice(choiceData)
    await loadAllNodes()
    isDirty.value = false
  }

  async function deleteChoice(choiceId: string) {
    await editorApi.deleteChoice(choiceId)
    await loadAllNodes()
  }

  function markDirty() {
    isDirty.value = true
  }

  return {
    nodes, choices, selectedNodeId, selectedChoiceId, isDirty,
    selectedNode, nodeChoices, nodesByType,
    loadAllNodes, selectNode, selectChoice,
    saveNode, deleteNode, saveChoice, deleteChoice,
    markDirty,
  }
})
```

#### 2.2.2 交互功能

| 交互 | 操作 | 说明 |
|------|------|------|
| 点击节点 | 选中 → InspectorPanel 显示属性 | `cy.on('select', 'node', ...)` |
| 双击节点 | 打开编辑弹窗 | 快速编辑名称/内容 |
| 右键节点 | 上下文菜单 | 新增/删除/复制/设为特殊 |
| 拖拽连线 | 从源节点拖到目标节点 | 创建新 Choice |
| 拖拽节点 | 调整节点位置 | 更新 position 字段 |

| 框选 | 多选节点 | 批量操作（暂不实现） |

#### 2.2.3 环形参考线

在画布上绘制固定参考线，表示莫比乌斯环的 200 单位结构：

- 外圈虚线圆（半径 R）
- 8 个主节点固定位置标记（A=0°, B=45°, C=90°...）
- A↔E 莫比乌斯扭转示意线（直径方向的虚线，加 180° 扭转符号）

#### 2.2.4 连线创建流程

```
1. 用户在源节点上 mousedown
2. 出现临时的"橡皮筋"线跟随鼠标
3. 鼠标移动到目标节点上释放
4. 弹出 ChoiceQuickCreate 弹窗（仅需填写 text 和 effects）
5. 确认后创建 Choice 并渲染边
```

### 2.6 编辑器特色功能（Day 19-20）

#### 2.6.1 实时预览

编辑器顶部"预览"按钮，使用 `window.open('/play?preview_node=A')` 打开播放器窗口，以当前编辑状态预览游戏效果。

#### 2.6.2 导出/导入

- 导出：将所有节点和选择导出为 JSON 文件（下载到本地）
- 导入：上传 JSON 文件覆盖当前数据（需确认）

#### 2.6.3 条件语法实时校验

输入条件表达式时实时调用 `check_syntax()` 显示 ✓ 或 ✗。

#### 2.6.4 环形布局验证

自动检查以下约束：
- 主节点 position 是否为 0/25/50/75/100/125/150/175
- 是否存在死节点（no incoming edges, no outgoing edges）
- 是否存在孤立节点组
- 从 A 出发是否可到达所有主节点
