### 2.1 导入脚本完善（Day 1-4）

#### 2.1.1 完整导入流程

```python
# backend/import_story.py
import json
import os
from pathlib import Path
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.story import StoryNode, Choice
from app.models.save import Item, FlagDef, NPC, PlayerAttributeDef
from app.engine.condition_eval import ConditionEvaluator

class StoryImporter:
    def __init__(self, data_dir: str = "story_data"):
        self.data_dir = Path(data_dir)
        self.errors = []
        self.warnings = []

    async def import_all(self, session: AsyncSession) -> bool:
        """批量导入，返回 True 表示全成功"""
        # 清除旧数据
        await self._clear_story_data(session)

        # 逐类导入
        await self._import_meta(session)
        await self._import_flags(session)
        await self._import_items(session)
        await self._import_npcs(session)
        await self._import_player_attributes(session)
        await self._import_nodes(session)
        await self._import_choices(session)

        # 校验
        await self._validate(session)

        if self.errors:
            print(f"\n❌ 导入失败: {len(self.errors)} 个错误")
            for err in self.errors:
                print(f"   - {err}")
            return False

        if self.warnings:
            print(f"\n⚠ 导入完成但有 {len(self.warnings)} 个警告:")
            for warn in self.warnings:
                print(f"   - {warn}")

        print("✅ 全部数据导入成功")
        return True

    def _read_json(self, filename: str) -> dict | list:
        path = self.data_dir / filename
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)

    async def _validate(self, session: AsyncSession):
        """校验外键完整性和条件语法"""
        # 1. 所有 choice.next_node_id 必须指向存在的 story_nodes.id
        # 2. 所有 choice.condition 必须通过 check_syntax()
        # 3. 所有 choice.effects 中的道具/flag/属性 ID 必须已定义
        # 4. 主节点 position 必须是 [0, 25, 50, 75, 100, 125, 150, 175, -1]
        # 5. cycle_variants key 必须匹配 cycle_N 或 cycle_N+
        # 6. cross_surface=true 的道具必须在 A 或 E 的 choice 中被引用
```

#### 2.1.2 CLI 入口

```python
# 命令行调用
# python -m backend.import_story --data-dir story_data --db cycle_master.db

import asyncio
import argparse

async def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--data-dir', default='story_data')
    parser.add_argument('--db-url', default='sqlite+aiosqlite:///cycle_master.db')
    parser.add_argument('--dry-run', action='store_true', help='仅校验不写入')
    args = parser.parse_args()

    importer = StoryImporter(args.data_dir)
    if args.dry_run:
        await importer.validate_only()
    else:
        success = await importer.import_all()
        exit(0 if success else 1)

asyncio.run(main())
```

#### 2.2.3 节点创建方式选择

**方式一：通过编辑器手动创建（推荐 P0-P1）**
- 8 主节点 + 2 特殊节点在编辑器中逐个创建
- 优点：可视化，所见即所得
- 缺点：纯文本量大

**方式二：手写 JSON + 导入（推荐 P2-P4）**
- 使用 VSCode 编写 JSON 文件
- 运行 `import_story.py` 导入
- 优点：批量高效，可版本控制
- 缺点：需要熟悉 JSON 格式

**方式三：混合（实际采用）**
- 结构（节点/选择关系）用编辑器创建
- 内容（长文本）用手写 JSON 批量导入
