#### 1.2.2 音频管理器

```typescript
// frontend/src/composables/useAudio.ts
import { ref } from 'vue'

const audioContext = new (window.AudioContext || (window as any).webkitAudioContext)()

export function useAudio() {
  const bgmVolume = ref(0.3)
  const sfxVolume = ref(0.6)
  const isMuted = ref(false)

  let currentBGM: HTMLAudioElement | null = null

  function playSFX(name: string) {
    if (isMuted.value) return
    const audio = new Audio(`/assets/audio/sfx/${name}.mp3`)
    audio.volume = sfxVolume.value
    audio.play().catch(() => {})  // 忽略自动播放限制
  }

  function playBGM(nodeId: string) {
    // 节点切换时交叉淡入淡出 BGM
    if (currentBGM) {
      fadeOut(currentBGM, 1000)
    }
    const audio = new Audio(`/assets/audio/bgm/${nodeId}.mp3`)
    audio.volume = 0
    audio.loop = true
    audio.play()
    fadeIn(audio, bgmVolume.value, 2000)
    currentBGM = audio
  }

  function fadeIn(audio: HTMLAudioElement, target: number, duration: number) { /* ... */ }
  function fadeOut(audio: HTMLAudioElement, duration: number) { /* ... */ }

  return { playSFX, playBGM, bgmVolume, sfxVolume, isMuted }
}
```

#### 1.5.2 后端嵌入方案

Tauri 桌面应用需要后端同时运行。两种方案：

**方案 A：独立进程（推荐）**
- 后端作为独立 Python 进程启动
- Tauri 启动时自动 `spawn` Python 后端
- 用户无感知，但需要打包 Python runtime

**方案 B：PyO3 嵌入**
- 使用 PyO3 将 Python 引擎嵌入 Rust
- 但 FastAPI 需要独立 ASGI server，嵌入复杂度高
- 仅嵌入图引擎核心（去掉 HTTP 层）
- 前端通过 Tauri command 调用

**实际采用方案 A**：使用 PyInstaller 将后端打包为独立 exe，Tauri 启动时自动拉起。

```rust
// src-tauri/src/main.rs
use std::process::Command;

fn main() {
    tauri::Builder::default()
        .setup(|app| {
            // 启动 Python 后端
            let backend = Command::new("backend/cycle_master_backend.exe")
                .arg("--port")
                .arg("8000")
                .spawn()
                .expect("Failed to start backend");

            // 等待后端就绪
            std::thread::sleep(std::time::Duration::from_secs(2));
            Ok(())
        })
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
```

#### 1.5.3 打包步骤

```bash
# 1. 打包 Python 后端
cd backend
pyinstaller --onefile --name cycle_master_backend app/main.py

# 2. 构建前端
cd frontend
npm run build

# 3. Tauri 打包
npx tauri build
```
